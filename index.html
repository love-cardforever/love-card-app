async function handleRegister(e) { e.preventDefault(); const name = document.getElementById('reg-name').value.trim(); const gender = document.querySelector('input[name="reg-gender"]:checked').value; const birthyear = document.getElementById('reg-birthyear').value.trim(); const password = document.getElementById('reg-pass').value.trim(); const email = document.getElementById('reg-email').value.trim(); let genderPrefix = "12"; if (gender === 'male') genderPrefix = "11"; else if (gender === 'female') genderPrefix = "10"; const db = await getDB(); if (!db.users) db.users = []; // ระบบตรวจสอบและสุ่มรหัสที่ไม่ซ้ำกับคนอื่นในระบบแน่นอน let userID = ""; let isDuplicate = true; let safetyCounter = 0; const birthYearLastDigit = birthyear.slice(-1); while (isDuplicate && safetyCounter < 50) { let counter = db.globalUserCounter || 0; let groupChar = db.globalGroupChar || 'A'; let countStr = String(counter).padStart(3, '0'); userID = `${genderPrefix}${birthYearLastDigit}${countStr}${groupChar}`; // เช็คว่า User ID นี้มีอยู่แล้วหรือยัง const existingUser = db.users.find(u => u.id === userID); if (!existingUser) { isDuplicate = false; } // ขยับ counter เตรียมไว้สำหรับคนถัดไป db.globalUserCounter = (counter + 1); if (db.globalUserCounter > 999) { db.globalUserCounter = 0; let charCode = groupChar.charCodeAt(0) + 1; db.globalGroupChar = String.fromCharCode(charCode); } safetyCounter++; } pendingUserRegData = { id: userID, name, gender, birthyear, password, email, coupleId: null, createdTime: new Date().toISOString() }; realOtpCode = String(Math.floor(100000 + Math.random() * 900000)); const btn = document.getElementById('btn-submit-reg'); btn.innerText = "กำลังส่งอีเมล OTP..."; btn.disabled = true; try { await emailjs.send("service_default", "template_default", { to_email: email, to_name: name, otp_code: realOtpCode }); document.getElementById('modal-otp').classList.remove('hidden'); } catch (err) { document.getElementById('modal-otp').classList.remove('hidden'); showCustomAlert('แจ้งเตือนการส่งอีเมล', `ไม่สามารถเชื่อมต่อเซิร์ฟเวอร์เมลได้ รหัส OTP ของคุณคือ: ${realOtpCode} (ติดต่อ IG: @mumu_apr200x)`); } finally { btn.innerText = "ส่งรหัส OTP ไปยังอีเมล"; btn.disabled = false; } }
